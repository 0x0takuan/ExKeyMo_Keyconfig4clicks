type OVERLAY

# 1. 物理キーを基本キーコードにマッピング
# iOSモードのCtrlキー：CTRL
map key 240 CTRL_LEFT

# Clicksキー：全角半角
map key 15  ZENKAKU_HANKAKU

# Keyboardキー：カーソル左
map key 56  DPAD_LEFT

# iOSモードのKeyboardキー：カーソル左
map key 374 DPAD_LEFT

# Voiceキー：カーソル右
map key 582 DPAD_RIGHT

# .キー：PERIOD
map key 52  PERIOD

# $キー：4
map key 5   4

# 2. キーの動作定義
key PERIOD {
    label:         '.'
    base:          '-'
}

key 4 {
    label:         '4'
    base:          '4'
    shift:         '.'
}
